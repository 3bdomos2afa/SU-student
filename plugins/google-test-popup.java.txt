// SU_GOOGLE_TEST_POPUP_V1: isolated trial only; no credential/form/Google-token bridge.
    private android.app.Dialog suGoogleDialog;

    private static boolean suAllowedGoogleTestUrl(String value) {
        try {
            android.net.Uri u = android.net.Uri.parse(value);
            if (u.getUserInfo() != null) return false;
            int port = u.getPort();
            boolean google = "https".equals(u.getScheme())
                && "accounts.google.com".equals(u.getHost()) && (port == -1 || port == 443);
            boolean portal = "http".equals(u.getScheme())
                && "unicodesis.su.edu.eg".equals(u.getHost()) && (port == -1 || port == 80);
            return google || portal;
        } catch (Exception ignored) { return false; }
    }

    private boolean suOpenGoogleTestPopup(final WebView parent, Message message) {
        Activity activity = mWebView.getThemedReactContext().getCurrentActivity();
        if (activity == null || activity.isFinishing()) return false;
        if (suGoogleDialog != null) suGoogleDialog.dismiss();

        final android.app.Dialog dialog = new android.app.Dialog(activity);
        suGoogleDialog = dialog;
        final android.widget.LinearLayout layout = new android.widget.LinearLayout(activity);
        layout.setOrientation(android.widget.LinearLayout.VERTICAL);
        layout.setBackgroundColor(android.graphics.Color.WHITE);
        final android.widget.LinearLayout header = new android.widget.LinearLayout(activity);
        final android.widget.TextView address = new android.widget.TextView(activity);
        address.setText("Google sign-in TEST — accounts.google.com");
        address.setTextColor(android.graphics.Color.BLACK);
        address.setPadding(16, 16, 8, 16);
        header.addView(address, new android.widget.LinearLayout.LayoutParams(0,
            android.view.ViewGroup.LayoutParams.WRAP_CONTENT, 1));
        android.widget.Button close = new android.widget.Button(activity);
        close.setText("Close");
        close.setOnClickListener(v -> dialog.dismiss());
        header.addView(close);
        layout.addView(header);
        final WebView child = new WebView(activity);
        android.webkit.WebSettings settings = child.getSettings();
        settings.setJavaScriptEnabled(true);
        settings.setDomStorageEnabled(true);
        settings.setUserAgentString(parent.getSettings().getUserAgentString());
        settings.setAllowFileAccess(false);
        settings.setAllowContentAccess(false);
        settings.setMixedContentMode(android.webkit.WebSettings.MIXED_CONTENT_NEVER_ALLOW);
        settings.setSaveFormData(false);
        settings.setSupportMultipleWindows(true);
        settings.setJavaScriptCanOpenWindowsAutomatically(false);
        android.webkit.CookieManager.getInstance().setAcceptThirdPartyCookies(child, true);
        // Plain WebView: intentionally NO addJavascriptInterface or injected script.
        child.setWebViewClient(new WebViewClient() {
            @Override
            public boolean shouldOverrideUrlLoading(WebView w, android.webkit.WebResourceRequest request) {
                return !suAllowedGoogleTestUrl(request.getUrl().toString());
            }
            @Override
            public boolean shouldOverrideUrlLoading(WebView w, String url) {
                return !suAllowedGoogleTestUrl(url);
            }
            @Override
            public void onPageStarted(WebView w, String url, android.graphics.Bitmap icon) {
                if (!suAllowedGoogleTestUrl(url)) {
                    w.stopLoading();
                    address.setText("Blocked navigation — close this test");
                    return;
                }
                android.net.Uri u = android.net.Uri.parse(url);
                address.setText("TEST — " + u.getScheme() + "://" + u.getHost());
            }
            @Override
            public void onReceivedSslError(WebView w, android.webkit.SslErrorHandler handler,
                android.net.http.SslError error) {
                handler.cancel();
                address.setText("Certificate error — sign-in stopped");
            }
        });
        child.setWebChromeClient(new WebChromeClient() {
            @Override
            public void onCloseWindow(WebView window) { dialog.dismiss(); }
            @Override
            public void onPermissionRequest(android.webkit.PermissionRequest request) { request.deny(); }
            @Override
            public void onGeolocationPermissionsShowPrompt(String origin,
                android.webkit.GeolocationPermissions.Callback callback) {
                callback.invoke(origin, false, false);
            }
        });
        final View.OnAttachStateChangeListener detach = new View.OnAttachStateChangeListener() {
            @Override public void onViewAttachedToWindow(View v) { }
            @Override public void onViewDetachedFromWindow(View v) { dialog.dismiss(); }
        };
        parent.addOnAttachStateChangeListener(detach);
        dialog.setOnDismissListener(ignored -> {
            parent.removeOnAttachStateChangeListener(detach);
            if (suGoogleDialog == dialog) suGoogleDialog = null;
            child.stopLoading();
            layout.removeView(child);
            child.destroy();
        });
        layout.addView(child, new android.widget.LinearLayout.LayoutParams(
            android.view.ViewGroup.LayoutParams.MATCH_PARENT, 0, 1));
        dialog.setContentView(layout);
        dialog.show();
        if (dialog.getWindow() != null) dialog.getWindow().setLayout(
            android.view.ViewGroup.LayoutParams.MATCH_PARENT,
            android.view.ViewGroup.LayoutParams.MATCH_PARENT);
        final WebView.WebViewTransport transport = (WebView.WebViewTransport) message.obj;
        transport.setWebView(child);
        message.sendToTarget();
        return true;
    }
