package __PACKAGE__

import android.app.Activity
import android.content.ClipData
import android.content.ClipboardManager
import android.content.Context
import android.content.Intent
import android.graphics.Color
import android.graphics.Typeface
import android.os.Bundle
import android.view.ViewGroup
import android.widget.Button
import android.widget.LinearLayout
import android.widget.ScrollView
import android.widget.TextView
import android.widget.Toast
import java.io.File

// Plain Android screen shown in the ":crash" process. Does not load React Native.
class CrashActivity : Activity() {
  companion object { const val EXTRA_REPORT = "report" }

  override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)
    val report = intent.getStringExtra(EXTRA_REPORT) ?: run {
      val f = File(filesDir, CrashReporter.REPORT_FILE)
      if (f.exists()) f.readText() else "No report saved."
    }
    val density = resources.displayMetrics.density
    val pad = (16 * density).toInt()
    val root = LinearLayout(this)
    root.orientation = LinearLayout.VERTICAL
    root.setBackgroundColor(Color.WHITE)
    root.setPadding(pad, pad * 2, pad, pad)

    val title = TextView(this)
    title.text = "التطبيق وقف قبل ما يفتح"
    title.textSize = 20f
    title.setTypeface(null, Typeface.BOLD)
    title.setTextColor(Color.BLACK)
    root.addView(title)

    val hint = TextView(this)
    hint.text = "ده تقرير الخطأ من الموبايل. اضغط «مشاركة» أو «نسخ» وابعته للمطور. مفيش فيه أي بيانات دخول."
    hint.textSize = 14f
    hint.setTextColor(Color.DKGRAY)
    hint.setPadding(0, pad / 2, 0, pad / 2)
    root.addView(hint)

    val scroll = ScrollView(this)
    val body = TextView(this)
    body.text = report
    body.typeface = Typeface.MONOSPACE
    body.textSize = 11f
    body.setTextColor(Color.BLACK)
    body.setTextIsSelectable(true)
    body.textDirection = TextView.TEXT_DIRECTION_LTR
    body.setPadding(pad / 2, pad / 2, pad / 2, pad / 2)
    body.setBackgroundColor(Color.rgb(243, 244, 246))
    scroll.addView(body)
    root.addView(scroll, LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, 0, 1f))

    val buttons = LinearLayout(this)
    buttons.orientation = LinearLayout.HORIZONTAL
    buttons.setPadding(0, pad / 2, 0, 0)
    val share = Button(this)
    share.text = "مشاركة"
    share.setOnClickListener {
      val send = Intent(Intent.ACTION_SEND)
      send.type = "text/plain"
      send.putExtra(Intent.EXTRA_SUBJECT, "SU Student crash report")
      send.putExtra(Intent.EXTRA_TEXT, report)
      startActivity(Intent.createChooser(send, "ابعت التقرير"))
    }
    val copy = Button(this)
    copy.text = "نسخ"
    copy.setOnClickListener {
      val cm = getSystemService(Context.CLIPBOARD_SERVICE) as ClipboardManager
      cm.setPrimaryClip(ClipData.newPlainText("SU Student crash report", report))
      Toast.makeText(this, "اتنسخ", Toast.LENGTH_SHORT).show()
    }
    val retry = Button(this)
    retry.text = "افتح التطبيق"
    retry.setOnClickListener {
      File(filesDir, "su-start-count").delete()
      val launch = Intent(this, MainActivity::class.java)
      launch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK)
      startActivity(launch)
      finish()
    }
    val lp = LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f)
    buttons.addView(share, lp)
    buttons.addView(copy, lp)
    buttons.addView(retry, lp)
    root.addView(buttons)
    setContentView(root)
  }
}
